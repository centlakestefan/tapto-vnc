# For tools/make-movie.py, which assembles a --screenshots directory into an
# mp4. Nothing else in this repository needs Python: the program itself is C++
# and builds with CMake alone.
#
# PyAV is FFmpeg's own libraries with a Python binding, and its wheels carry
# those libraries with them — so this line is the whole install, with no system
# ffmpeg, no PATH to arrange, and the same encoder behaving the same way on
# every machine.
#
#     pip install -r tools/requirements.txt
av>=13.0

# Pillow draws the panel beside the screen — the task and the model's own
# reasoning, which frames.jsonl records but a frame cannot show. It also loads
# and composes the frames, which is why there is no FFmpeg filter graph here.
pillow>=10.0
